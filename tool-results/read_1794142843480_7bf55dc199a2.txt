    27→    27→interface TokenState {

    32→    32→
    33→    33→interface QrSession {
    34→    34→  state: string

    36→    36→  qrImg: string | null
    37→    37→  /** 服务端连通性预检：微信开放平台 / 哲风壁纸是否可达 */

    43→    43→

    45→    45→
    46→    46→/** 令牌有效性验证状态（用官网接口实测，避免「令牌已失效却显示有效」） */

    49→    49→const QR_TTL = 170

    52→    52→/** 轮询连续失败阈值：达到即停止并报错（约 6 秒内给出明确反馈，杜绝无限等待） */
    53→    53→const POLL_FAIL_LIMIT = 3
    54→    54→
    55→    55→/**
    56→    56→ * 登录令牌设置：
    57→    57→ * - 扫码登录：直显服务端提取的「微信开放平台扫码二维码」图片（免 iframe 跨域拦截），
    58→    58→ *   状态区分「等待扫码 → 已扫码待确认 → 成功」，扫码确认后令牌自动入库
    59→    59→ * - 手动粘贴：从官网浏览器开发者工具复制令牌
    60→    60→ * 令牌用于下载原图（与官网分辨率、文件大小完全一致），仅保存在本地数据库。
    61→    61→ *
    62→    62→ * 扫码等待体验优化（避免「登录不上一直等待」）：
    63→    63→ * - 发起时服务端预检微信 / 哲风连通性，不可达直接报错并给手动粘贴兜底
    64→    64→ * - 二维码图片 12 秒未加载出来 → 判定加载失败
    65→    65→ * - 轮询连续 3 次失败 → 停止轮询并报错
    66→    66→ * - 倒计时归零 → 本地立即判过期（不等源站）
    67→    67→ * - 等待期间随时可刷新二维码 / 切换手动粘贴
    68→    68→ */
    69→    69→export function TokenSettingsDialog({
    70→    70→  open,
    71→    71→  onOpenChange,
    72→    72→  probeWtId,
    73→    73→}: {
    74→    74→  open: boolean
    75→    75→  onOpenChange: (open: boolean) => void
    76→    76→  /** 用于验证令牌有效性的壁纸 id（当前列表任一张，官网接口需要） */
    77→    77→  probeWtId?: string | null
    78→    78→}) {
    79→    79→  const [state, setState] = useState<TokenState>({ token: null, source: null, createdAt: null })
    80→    80→  const [mode, setMode] = useState<"qrcode" | "manual">("qrcode")
    81→    81→  const [manualInput, setManualInput] = useState("")
    82→    82→  const [saving, setSaving] = useState(false)
    83→    83→  /** 令牌有效性（官网实测）：checking / valid / invalid / unknown */
    84→    84→  const [verify, setVerify] = useState<VerifyState>("idle")
    85→    85→  /** 失效原因（invalid 时展示） */
    86→    86→  const [verifyMsg, setVerifyMsg] = useState<string | null>(null)
    87→    87→
    88→    88→  // 扫码会话
    89→    89→  const [qr, setQr] = useState<QrSession | null>(null)
    90→    90→  const [phase, setPhase] = useState<QrPhase>("loading")
    91→    91→  /** 具体失败原因（error 阶段展示） */
    92→    92→  const [errorMsg, setErrorMsg] = useState<string | null>(null)
    93→    93→  const [countdown, setCountdown] = useState(QR_TTL)
    94→    94→  const [qrNonce, setQrNonce] = useState(0)
    95→    95→  /** 已扫码阶段的服务端过程提示（如「授权已确认，正在获取令牌」） */
    96→    96→  const [scannedNote, setScannedNote] = useState<string | null>(null)
    97→    97→  /** 二维码 iframe 是否已触发 load 事件 */
    98→    98→  const qrLoadedRef = useRef(false)
    99→    99→  /** 轮询连续失败计数 */
   100→   100→  const pollFailRef = useRef(0)
   101→   101→
   102→   102→  const refresh = useCallback(async () => {
   103→   103→    try {
   104→   104→      const res = await fetch("/api/token", { cache: "no-store" })
   105→   105→      if (res.ok) setState(await res.json())
   106→   106→    } catch {
   107→   107→      /* 静默失败，保留当前展示 */
   108→   108→    }
   109→   109→  }, [])
   110→   110→
   111→   111→  // 验证令牌有效性：弹窗打开 / 令牌变化（重新扫码入库后）时，用官网接口实测一次
   112→   112→  const runVerify = useCallback(async (wtId: string) => {
   113→   113→    setVerify("checking")
   114→   114→    setVerifyMsg(null)
   115→   115→    try {
   116→   116→      const res = await fetch(`/api/token/verify?wtId=${encodeURIComponent(wtId)}`, {
   117→   117→        cache: "no-store",
   118→   118→      })
   119→   119→      const data = (await res.json().catch(() => null)) as { valid?: boolean | null; msg?: string } | null
   120→   120→      if (data?.valid === true) setVerify("valid")
   121→   121→      else if (data?.valid === false) {
   122→   122→        setVerify("invalid")
   123→   123→        setVerifyMsg(typeof data.msg === "string" ? data.msg : null)
   124→   124→      } else setVerify("unknown")
   125→   125→    } catch {
   126→   126→      setVerify("unknown")
   127→   127→    }
   128→   128→  }, [])
   129→   129→
   130→   130→  useEffect(() => {
   131→   131→    if (!open || !state.token || !probeWtId) return
   132→   132→    let cancelled = false
   133→   133→    // 微任务中执行，避免 effect 内同步 setState
   134→   134→    Promise.resolve().then(() => {
   135→   135→      if (cancelled) return
   136→   136→      void runVerify(probeWtId)
   137→   137→    })
   138→   138→    return () => {
   139→   139→      cancelled = true
   140→   140→    }
   141→   141→  }, [open, state.token, probeWtId, runVerify])
   142→   142→
   143→   143→  // 打开时重置 + 拉取令牌状态（微任务中执行，避免 effect 内同步 setState）
   144→   144→  useEffect(() => {
   145→   145→    if (!open) return
   146→   146→    let cancelled = false
   147→   147→    Promise.resolve().then(() => {
   148→   148→      if (cancelled) return
   149→   149→      void refresh()
   150→   150→      setMode("qrcode")
   151→   151→      setManualInput("")
   152→   152→      setQr(null)
   153→   153→      setPhase("loading")
   154→   154→      setErrorMsg(null)
   155→   155→      setScannedNote(null)
   156→   156→      setCountdown(QR_TTL)
   157→   157→      setVerify("idle")
   158→   158→      setVerifyMsg(null)
   159→   159→      qrLoadedRef.current = false
   160→   160→      pollFailRef.current = 0
   161→   161→    })
   162→   162→    return () => {
   163→   163→      cancelled = true
   164→   164→    }
   165→   165→  }, [open, refresh])
   166→   166→
   167→   167→  // 发起扫码会话（打开弹窗 / 点击刷新时；微任务中执行，避免 effect 内同步 setState）
   168→   168→  useEffect(() => {
   169→   169→    if (!open || mode !== "qrcode") return
   170→   170→    let cancelled = false
   171→   171→    Promise.resolve().then(() => {
   172→   172→      if (cancelled) return
   173→   173→      setPhase("loading")
   174→   174→      setCountdown(QR_TTL)
   175→   175→      setErrorMsg(null)
   176→   176→      setScannedNote(null)
   177→   177→      qrLoadedRef.current = false
   178→   178→      pollFailRef.current = 0
   179→   179→      void (async () => {
   180→   180→        try {
   181→   181→          const res = await fetch("/api/zf-login", { method: "POST" })
   182→   182→          if (!res.ok) throw new Error()
   183→   183→          const data: QrSession = await res.json()
   184→   184→          if (cancelled) return
   185→   185→          // 连通性预检：不可达时直接给出明确错误（不进入无限等待）
   186→   186→          if (data.wxReachable === false) {
   187→   187→            setQr(null)
   188→   188→            setPhase("error")
   189→   189→            setErrorMsg(
   190→   190→              data.zfReachable === false
   191→   191→                ? "无法访问微信开放平台与哲风壁纸服务，请检查网络后重试"
   192→   192→                : "无法访问微信开放平台（open.weixin.qq.com），二维码显示不出来"
   193→   193→            )
   194→   194→            return
   195→   195→          }
   196→   196→          if (data.zfReachable === false) {
   197→   197→            setQr(null)
   198→   198→            setPhase("error")
   199→   199→            setErrorMsg("无法访问哲风壁纸登录服务（haowallpaper.com），扫码后也拿不到令牌")
   200→   200→            return
   201→   201→          }
   202→   202→          if (!data.qrImg) {
   203→   203→            // 服务端提取二维码失败（qrconnect 页面抓取/解析失败）
   204→   204→            setQr(null)
   205→   205→            setPhase("error")
   206→   206→            setErrorMsg(data.startError ?? "二维码会话创建失败，请刷新重试或改用手动粘贴")
   207→   207→            return
   208→   208→          }
   209→   209→          setQr(data)
   210→   210→          setPhase("waiting")
   211→   211→        } catch {
   212→   212→          if (!cancelled) {
   213→   213→            setQr(null)
   214→   214→            setPhase("error")
   215→   215→            setErrorMsg("二维码会话创建失败，请检查网络后重试")
   216→   216→          }
   217→   217→        }
   218→   218→      })()
   219→   219→    })
   220→   220→    return () => {
   221→   221→      cancelled = true
   222→   222→    }
   223→   223→  }, [open, mode, qrNonce])
   224→   224→
   225→   225→  // 倒计时（waiting 与 scanned 阶段都计时）
   226→   226→  useEffect(() => {
   227→   227→    if (!open || mode !== "qrcode" || (phase !== "waiting" && phase !== "scanned")) return
   228→   228→    const t = setInterval(() => setCountdown((c) => (c > 0 ? c - 1 : 0)), 1000)
   229→   229→    return () => clearInterval(t)
   230→   230→  }, [open, mode, phase])
   231→   231→
   232→   232→  // 倒计时归零 → 本地立即判过期并停止轮询（不等源站，杜绝无限等待）
   233→   233→  useEffect(() => {
   234→   234→    if (!open || mode !== "qrcode" || (phase !== "waiting" && phase !== "scanned") || countdown > 0)
   235→   235→      return
   236→   236→    Promise.resolve().then(() => {
   237→   237→      setPhase("expired")
   238→   238→    })
   239→   239→  }, [open, mode, phase, countdown])
   240→   240→
   241→   241→  // 二维码图片加载超时检测：超时判定加载失败（而不是一直「等待扫码」）
   242→   242→  useEffect(() => {
   243→   243→    if (!open || mode !== "qrcode" || (phase !== "waiting" && phase !== "scanned") || !qr) return
   244→   244→    const t = setTimeout(() => {
   245→   245→      if (!qrLoadedRef.current) {
   246→   246→        setPhase("error")
   247→   247→        setErrorMsg("二维码图片加载超时，请刷新重试或改用手动粘贴")
   248→   248→      }
   249→   249→    }, QR_LOAD_TIMEOUT)
   250→   250→    return () => clearTimeout(t)
   251→   251→  }, [open, mode, phase, qr])
   252→   252→
   253→   253→  // 轮询扫码状态（服务端后台已在长轮询微信，这里每 2 秒查一次会话状态）
   254→   254→  useEffect(() => {
   255→   255→    if (!open || mode !== "qrcode" || (phase !== "waiting" && phase !== "scanned") || !qr) return
   256→   256→    let stopped = false
   257→   257→    const timer = setInterval(
   258→   258→      () => {
   259→   259→        if (stopped) return
   260→   260→        void (async () => {
   261→   261→          try {
   262→   262→            const res = await fetch(`/api/zf-login?state=${encodeURIComponent(qr.state)}`, {
   263→   263→              cache: "no-store",
   264→   264→            })
   265→   265→            const data = await res.json().catch(() => null)
   266→   266→            if (stopped || !data) return
   267→   267→            if (data.status === "ok") {
   268→   268→              stopped = true
   269→   269→              pollFailRef.current = 0
   270→   270→              setPhase("ok")
   271→   271→              toast.success("扫码登录成功，令牌已自动保存")
   272→   272→              void refresh()
   273→   273→            } else if (data.status === "scanned") {
   274→   274→              // 已扫码（或已确认授权，服务端正在换取令牌）
   275→   275→              pollFailRef.current = 0
   276→   276→              setPhase("scanned")
   277→   277→              if (typeof data.msg === "string" && data.msg) setScannedNote(data.msg)
   278→   278→            } else if (data.status === "expired") {
   279→   279→              stopped = true
   280→   280→              setPhase("expired")
   281→   281→            } else if (data.status === "pending") {
   282→   282→              pollFailRef.current = 0
   283→   283→              // 从「已扫码」回落（用户取消授权等）
   284→   284→              setPhase("waiting")
   285→   285→            } else {
   286→   286→              // error → 计数；连续多次失败才停止（单次抖动继续轮询）
   287→   287→              pollFailRef.current += 1
   288→   288→              if (pollFailRef.current >= POLL_FAIL_LIMIT) {
   289→   289→                stopped = true
   290→   290→                setPhase("error")
   291→   291→                setErrorMsg(
   292→   292→                  typeof data.msg === "string" && data.msg
   293→   293→                    ? `登录状态查询失败：${data.msg}`
   294→   294→                    : "登录状态查询连续失败，请检查网络后重试"
   295→   295→                )
   296→   296→              }
   297→   297→            }
   298→   298→          } catch {
   299→   299→            pollFailRef.current += 1
   300→   300→            if (!stopped && pollFailRef.current >= POLL_FAIL_LIMIT) {
   301→   301→              stopped = true
   302→   302→              setPhase("error")
   303→   303→              setErrorMsg("网络异常，无法查询扫码状态，请检查网络后重试")
   304→   304→            }
   305→   305→          }
   306→   306→        })()
   307→   307→      },
   308→   308→      2000
   309→   309→    )
   310→   310→    return () => {
   311→   311→      stopped = true
   312→   312→      clearInterval(timer)
   313→   313→    }
   314→   314→  }, [open, mode, phase, qr, refresh])
   315→   315→
   316→   316→  const regenQr = () => {
   317→   317→    setQrNonce((n) => n + 1)
   318→   318→  }
   319→   319→
   320→   320→  const saveToken = async (token: string, source: "manual" | "qrcode") => {
   321→   321→    setSaving(true)
   322→   322→    try {
   323→   323→      const res = await fetch("/api/token", {
   324→   324→        method: "POST",
   325→   325→        headers: { "Content-Type": "application/json" },
   326→   326→        body: JSON.stringify({ token, source }),
   327→   327→      })
   328→   328→      const msg = await res.json().catch(() => null)
   329→   329→      if (!res.ok) throw new Error(msg?.error ?? "保存失败")
   330→   330→      setState(msg)
   331→   331→      toast.success("令牌已保存，下载原图功能已就绪")
   332→   332→    } catch (e) {
   333→   333→      toast.error(e instanceof Error ? e.message : "保存令牌失败")
   334→   334→    } finally {
   335→   335→      setSaving(false)
   336→   336→    }
   337→   337→  }
   338→   338→
   339→   339→  const deleteToken = async () => {
   340→   340→    try {
   341→   341→      const res = await fetch("/api/token", { method: "DELETE" })
   342→   342→      if (!res.ok) throw new Error()
   343→   343→      setState({ token: null, source: null, createdAt: null })
   344→   344→      toast.success("令牌已删除")
   345→   345→    } catch {
   346→   346→      toast.error("删除令牌失败")
   347→   347→    }
   348→   348→  }
   349→   349→
   350→   350→  const mm = Math.floor(countdown / 60)
   351→   351→  const ss = String(countdown % 60).padStart(2, "0")
   352→   352→
   353→   353→  return (
   354→   354→    <Dialog open={open} onOpenChange={onOpenChange}>
   355→   355→      <DialogContent className="max-w-md gap-0 overflow-hidden p-0 sm:max-w-md">
   356→   356→        <DialogHeader className="space-y-1.5 border-b bg-muted/40 px-5 py-4 text-left">
   357→   357→          <DialogTitle className="flex items-center gap-2 text-base font-bold">
   358→   358→            <KeyRound className="size-4 text-primary" />
   359→   359→            登录令牌设置
   360→   360→          </DialogTitle>
   361→   361→          <DialogDescription className="text-xs leading-relaxed">
   362→   362→            下载原图（与官网分辨率、文件大小完全一致）需要网站的登录令牌。推荐直接
   363→   363→            <span className="font-semibold text-foreground">微信扫码登录</span>
   364→   364→            自动获取，令牌仅保存在<span className="font-semibold text-foreground">本地数据库</span>。
   365→   365→          </DialogDescription>
   366→   366→        </DialogHeader>
   367→   367→
   368→   368→        <div className="custom-scrollbar max-h-[calc(100vh-7rem)] overflow-y-auto p-5">
   369→   369→          {/* 令牌状态卡（官网实测有效性，不再凭「入库了」就显示有效） */}
   370→   370→          {state.token ? (
   371→   371→            <div
   372→   372→              className={cn(
   373→   373→                "mb-4 rounded-xl border p-3.5",
   374→   374→                verify === "invalid" ? "border-amber-500/40 bg-amber-500/5" : "bg-muted/40"
   375→   375→              )}
   376→   376→            >
   377→   377→              <div className="flex items-center gap-2">
   378→   378→                {verify === "invalid" ? (
   379→   379→                  <ShieldX className="size-5 shrink-0 text-amber-500" />
   380→   380→                ) : (
   381→   381→                  <ShieldCheck
   382→   382→                    className={cn("size-5 shrink-0", verify === "valid" ? "text-emerald-500" : "text-muted-foreground")}
   383→   383→                  />
   384→   384→                )}
   385→   385→                <span className="text-sm font-bold">
   386→   386→                  {verify === "invalid" ? "令牌已失效" : "令牌有效"}
   387→   387→                  {verify === "checking" && " · 验证中…"}
   388→   388→                  {verify === "valid" && " · 已验证"}
   389→   389→                </span>
   390→   390→                <span className="rounded-full bg-muted px-2 py-0.5 font-mono text-[10px] text-muted-foreground">
   391→   391→                  {state.token}
   392→   392→                </span>
   393→   393→                {state.source === "qrcode" && (
   394→   394→                  <span className="rounded-full bg-primary/10 px-2 py-0.5 text-[10px] font-bold text-primary">
   395→   395→                    扫码登录
   396→   396→                  </span>
   397→   397→                )}
   398→   398→                {verify === "checking" && (
