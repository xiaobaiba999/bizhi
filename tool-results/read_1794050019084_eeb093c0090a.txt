     1→"use client"
     2→
     3→import { useCallback, useEffect, useRef, useState } from "react"
     4→import { motion, AnimatePresence } from "framer-motion"
     5→import { toast } from "sonner"
     6→import {
     7→  Check,
     8→  Download,
     9→  EyeOff,
    10→  Expand,
    11→  FolderOpen,
    12→  KeyRound,
    13→  RefreshCw,
    14→  Sparkles,
    15→  Trash2,
    16→  X,
    17→} from "lucide-react"
    18→import { cn } from "@/lib/utils"
    19→import { ZHEFENG_CATEGORIES, proxyImageUrl, type ZhefengWallpaper } from "@/lib/zhefeng"
    20→import {
    21→  fetchDownloadDir,
    22→  getDesktopApp,
    23→  onDesktopDownloadDone,
    24→  openDownloadFolder,
    25→  revealDownloadedFile,
    26→} from "@/lib/desktop"
    27→import { TokenSettingsDialog } from "@/components/workshop/token-settings"
    28→import { SelectionPanel } from "@/components/workshop/selection-panel"
    29→import {
    30→  Dialog,
    31→  DialogContent,
    32→  DialogHeader,
    33→  DialogTitle,
    34→} from "@/components/ui/dialog"
    35→
    36→interface ListState {
    37→  items: ZhefengWallpaper[]
    38→  page: number
    39→  hasMore: boolean
    40→  loading: boolean
    41→  error: string | null
    42→}
    43→
    44→const INITIAL: ListState = { items: [], page: 0, hasMore: true, loading: false, error: null }
    45→
    46→/** 下载原图代理链接（带 wtId：有令牌时走官网原图直链） */
    47→function downloadUrl(w: ZhefengWallpaper) {
    48→  return `/api/wallpapers/download?src=${encodeURIComponent(w.src)}&name=${encodeURIComponent(
    49→    w.title
    50→  )}&wtId=${encodeURIComponent(w.id)}`
    51→}
    52→
    53→/**
    54→ * 壁纸工坊主界面：壁纸数据实时来自哲风壁纸（haowallpaper.com）。
    55→ * 分类切换 + 无限滚动 + 已下载自动隐藏 + 多选批量下载。
    56→ */
    57→export function WorkshopTab() {
    58→  const [catKey, setCatKey] = useState("recommend")
    59→  const [reloadNonce, setReloadNonce] = useState(0)
    60→  const [list, setList] = useState<ListState>(INITIAL)
    61→  const [selected, setSelected] = useState<ZhefengWallpaper[]>([])
    62→  const [preview, setPreview] = useState<ZhefengWallpaper | null>(null)
    63→  const [previewLoading, setPreviewLoading] = useState(false)
    64→  const [tokenOpen, setTokenOpen] = useState(false)
    65→  /** 下载过的图片不再显示（默认开启，可临时关闭查看） */
    66→  const [hideDownloaded, setHideDownloaded] = useState(true)
    67→  const [downloadedCount, setDownloadedCount] = useState(0)
    68→
    69→  const scrollRef = useRef<HTMLDivElement>(null)
    70→  const sentinelRef = useRef<HTMLDivElement>(null)
    71→  const seqRef = useRef(0)
    72→  /** 桌面版下载保存目录（网页版为 null） */
    73→  const [dlDir, setDlDir] = useState<string | null>(null)
    74→  /** 下载完成提示防抖合并（批量下载 N 张 → 合并为一条「已保存 N 张」） */
    75→  const dlToastRef = useRef<{ count: number; lastAt: number; timer: ReturnType<typeof setTimeout> | null }>({
    76→    count: 0,
    77→    lastAt: 0,
    78→    timer: null,
    79→  })
    80→
    81→  // 桌面版：拉取下载目录（用于展示「图片存在的位置」）
    82→  useEffect(() => {
    83→    let alive = true
    84→    void fetchDownloadDir().then((dir) => {
    85→      if (alive && dir) setDlDir(dir)
    86→    })
    87→    return () => {
    88→      alive = false
    89→    }
    90→  }, [])
    91→
    92→  // 桌面版：下载落盘完成后提示保存位置，并可一键调起磁盘文件管理器
    93→  useEffect(() => {
    94→    return onDesktopDownloadDone(({ path, filename }) => {
    95→      const agg = dlToastRef.current
    96→      const now = Date.now()
    97→      // 距上次完成超过 10 秒 → 视为新一批下载，重新计数
    98→      if (now - agg.lastAt > 10_000) agg.count = 0
    99→      agg.lastAt = now
   100→      agg.count += 1
   101→      const total = agg.count
   102→      if (agg.timer) clearTimeout(agg.timer)
   103→      agg.timer = setTimeout(() => {
   104→        agg.timer = null
   105→        agg.count = 0
   106→        const dir =
   107→          path.length > filename.length && path.endsWith(filename)
   108→            ? path.slice(0, path.length - filename.length).replace(/[\\/]+$/, "")
   109→            : path
   110→        toast.success(total > 1 ? `已保存 ${total} 张壁纸到磁盘` : `壁纸已保存`, {
   111→          id: "wf-download-done",
   112→          description: `保存位置：${total > 1 ? dir : path}`,
   113→          action: {
   114→            label: "打开文件夹",
   115→            onClick: () => {
   116→              void (total > 1 ? openDownloadFolder() : revealDownloadedFile(path)).then((res) => {
   117→                if (!res.ok && res.error && res.error !== "not-desktop") {
   118→                  toast.error("打开文件夹失败", { description: res.error })
   119→                }
   120→              })
   121→            },
   122→          },
   123→          duration: 8000,
   124→        })
   125→      }, 700)
   126→    })
   127→  }, [])
   128→
   129→  const cat = ZHEFENG_CATEGORIES.find((c) => c.key === catKey) ?? ZHEFENG_CATEGORIES[0]
   130→
   131→  /** 拉取下一页壁纸（loading/hasMore 守卫 + seq 防过期响应 + 追加去重） */
   132→  const loadMore = useCallback(async () => {
   133→    if (list.loading || !list.hasMore || list.page === 0) return
   134→    const seq = ++seqRef.current
   135→    setList((s) => ({ ...s, loading: true, error: null }))
   136→
   137→    try {
   138→      const nextPage = list.page + 1
   139→      const res = await fetch(
   140→        `/api/wallpapers?cat=${catKey}&page=${nextPage}&hideDownloaded=${hideDownloaded ? 1 : 0}`,
   141→        { cache: "no-store" }
   142→      )
   143→      if (!res.ok) {
   144→        const msg = await res.json().catch(() => null)
   145→        throw new Error(msg?.error ?? "抓取失败")
   146→      }
   147→      const data: { items: ZhefengWallpaper[]; hasMore: boolean; downloadedCount?: number } =
   148→        await res.json()
   149→      if (seq !== seqRef.current) return
   150→
   151→      setList((s) => {
   152→        const known = new Set(s.items.map((i) => i.id))
   153→        return {
   154→          items: [...s.items, ...data.items.filter((i) => !known.has(i.id))],
   155→          page: nextPage,
   156→          hasMore: data.hasMore && data.items.length > 0,
   157→          loading: false,
   158→          error: null,
   159→        }
   160→      })
   161→      if (typeof data.downloadedCount === "number") setDownloadedCount(data.downloadedCount)
   162→    } catch (e) {
   163→      if (seq !== seqRef.current) return
   164→      setList((s) => ({ ...s, loading: false, error: e instanceof Error ? e.message : "抓取失败" }))
   165→    }
   166→  }, [catKey, list.page, list.loading, list.hasMore, hideDownloaded])
   167→
   168→  // 切换分类 / 手动重试 / 切换「隐藏已下载」时重置列表并重新抓取第一页
   169→  useEffect(() => {
   170→    seqRef.current++
   171→    const seq = seqRef.current
   172→    setList(INITIAL)
   173→    ;(async () => {
   174→      try {
   175→        const res = await fetch(
   176→          `/api/wallpapers?cat=${catKey}&page=1&hideDownloaded=${hideDownloaded ? 1 : 0}`,
   177→          { cache: "no-store" }
   178→        )
   179→        if (!res.ok) {
   180→          const msg = await res.json().catch(() => null)
   181→          throw new Error(msg?.error ?? "抓取失败")
   182→        }
   183→        const data: { items: ZhefengWallpaper[]; hasMore: boolean; downloadedCount?: number } =
   184→          await res.json()
   185→        if (seq !== seqRef.current) return
   186→        setList({
   187→          items: data.items,
   188→          page: 1,
   189→          hasMore: data.hasMore && data.items.length > 0,
   190→          loading: false,
   191→          error: null,
   192→        })
   193→        if (typeof data.downloadedCount === "number") setDownloadedCount(data.downloadedCount)
   194→      } catch (e) {
   195→        if (seq !== seqRef.current) return
   196→        setList({
   197→          items: [],
   198→          page: 0,
   199→          hasMore: true,
   200→          loading: false,
   201→          error: e instanceof Error ? e.message : "抓取失败",
   202→        })
   203→      }
   204→    })()
   205→  }, [catKey, reloadNonce, hideDownloaded])
   206→
   207→  // 无限滚动：哨兵进入可视区即加载下一页
   208→  useEffect(() => {
   209→    const el = sentinelRef.current
   210→    const root = scrollRef.current
   211→    if (!el || !root) return
   212→    const io = new IntersectionObserver(
   213→      (entries) => {
   214→        if (entries.some((en) => en.isIntersecting)) void loadMore()
   215→      },
   216→      { root, rootMargin: "400px 0px" }
   217→    )
   218→    io.observe(el)
   219→    return () => io.disconnect()
   220→  }, [loadMore])
   221→
   222→  const toggle = (w: ZhefengWallpaper) => {
   223→    setSelected((prev) => {
   224→      if (prev.some((p) => p.id === w.id)) return prev.filter((p) => p.id !== w.id)
   225→      if (prev.length >= 18) {
   226→        toast.error("最多选择 18 张")
   227→        return prev
   228→      }
   229→      return [...prev, w]
   230→    })
   231→  }
   232→
   233→  /** 触发一个下载（隐藏 iframe 导航方式，不弹新页） */
   234→  const triggerDownload = (w: ZhefengWallpaper) => {
   235→    const frame = document.createElement("iframe")
   236→    frame.style.display = "none"
   237→    frame.src = downloadUrl(w)
   238→    document.body.appendChild(frame)
   239→    setTimeout(() => frame.remove(), 120000)
   240→  }
   241→
   242→  /** 打开下载目录（桌面版调起磁盘文件管理器；网页版说明浏览器下载位置） */
   243→  const handleOpenDownloads = async () => {
   244→    if (getDesktopApp()) {
   245→      const r = await openDownloadFolder()
   246→      if (!r.ok) toast.error("打开下载文件夹失败", { description: r.error })
   247→      else toast.success("已打开下载文件夹", { description: dlDir ?? undefined, duration: 5000 })
   248→      return
   249→    }
   250→    toast.info("网页版下载由浏览器保存", {
   251→      description:
   252→        "默认在系统「下载」文件夹。安装桌面版后会自动整理到「下载\\壁纸工坊」，支持一键打开定位。",
   253→      duration: 6000,
   254→    })
   255→  }
   256→
   257→  /** 批量下载选中壁纸 */
   258→  const batchDownload = () => {
   259→    if (selected.length === 0) return
   260→    selected.forEach((w, i) => setTimeout(() => triggerDownload(w), i * 400))
   261→    toast.success(`开始下载 ${selected.length} 张壁纸，下载过的之后不再显示`, {
   262→      description: dlDir ? `保存位置：${dlDir}，完成后可点「打开文件夹」` : undefined,
   263→      duration: 6000,
   264→    })
   265→    // 乐观移除：稍后刷新列表让服务端记录生效
   266→    const ids = new Set(selected.map((w) => w.id))
   267→    setList((s) => ({ ...s, items: s.items.filter((w) => !ids.has(w.id)) }))
   268→    setSelected([])
   269→    if (hideDownloaded) setTimeout(() => setReloadNonce((n) => n + 1), 8000)
   270→  }
   271→
   272→  /** 单张下载：乐观移除（hideDownloaded 开启时） */
   273→  const onDownloadOne = (w: ZhefengWallpaper) => {
   274→    toast.success(`开始下载「${w.title}」，下载过的壁纸之后不再显示`)
   275→    if (hideDownloaded) {
   276→      setList((s) => ({ ...s, items: s.items.filter((i) => i.id !== w.id) }))
   277→      setDownloadedCount((n) => n + 1)
   278→    }
   279→  }
   280→
   281→  /** 清空下载记录（被隐藏的壁纸重新显示） */
   282→  const clearDownloads = async () => {
   283→    try {
   284→      const res = await fetch("/api/wallpapers/downloads", { method: "DELETE" })
   285→      if (!res.ok) throw new Error()
   286→      toast.success("下载记录已清空，所有壁纸恢复显示")
   287→      setReloadNonce((n) => n + 1)
   288→    } catch {
   289→      toast.error("清空下载记录失败")
   290→    }
   291→  }
   292→
   293→  const isPhone = cat.wpType === 2
   294→  const showSkeleton = list.page === 0 && (list.loading || !list.error)
   295→
   296→  return (
   297→    <section aria-label="壁纸工坊">
   298→      {/* 页头 */}
   299→      <div className="mb-5 flex flex-col gap-1.5">
   300→        <div className="flex flex-wrap items-center gap-2">
   301→          <h1 className="text-2xl font-black tracking-tight sm:text-[28px]">壁纸工坊</h1>
   302→          <span className="rounded-full border px-2 py-0.5 text-[10px] font-bold text-muted-foreground">
   303→            v1.1.1
   304→          </span>
   305→          <button
   306→            type="button"
   307→            onClick={() => setTokenOpen(true)}
   308→            className="inline-flex h-8 items-center gap-1.5 rounded-full border px-3 text-xs font-semibold text-muted-foreground transition-colors hover:border-primary/50 hover:text-foreground"
   309→          >
   310→            <KeyRound className="size-3.5" />
   311→            令牌设置
   312→          </button>
   313→          {/* 下载位置：桌面版直接调起磁盘文件夹；网页版说明浏览器默认下载目录 */}
   314→          <button
   315→            type="button"
   316→            onClick={() => void handleOpenDownloads()}
   317→            title={dlDir ?? "查看壁纸下载保存位置"}
   318→            aria-label={`打开壁纸下载文件夹${dlDir ? `：${dlDir}` : ""}`}
   319→            className="inline-flex h-8 items-center gap-1.5 rounded-full border px-3 text-xs font-semibold text-muted-foreground transition-colors hover:border-primary/50 hover:text-foreground"
   320→          >
   321→            <FolderOpen className="size-3.5" />
   322→            下载位置
   323→          </button>
   324→        </div>
   325→        <p className="flex items-center gap-1 text-sm text-muted-foreground">
   326→          <Sparkles className="size-3.5 shrink-0 text-primary" />
   327→          壁纸实时获取自
   328→          <a
   329→            href="https://haowallpaper.com"
   330→            target="_blank"
   331→            rel="noreferrer"
   332→            className="font-semibold text-foreground underline decoration-primary/40 underline-offset-2 hover:decoration-primary"
   333→          >
   334→            哲风壁纸
   335→          </a>
   336→          ，下载原图，下载过的自动隐藏
   337→        </p>
   338→      </div>
   339→
   340→      {/* 主体：左侧壁纸区 + 右侧已选面板（md 起常驻显示，避免窄断点看不到） */}
   341→      <div className="flex flex-col gap-5 md:flex-row">
   342→        <div className="min-w-0 flex-1">
   343→          {/* 分类筛选 + 已下载过滤（容器内吸顶） */}
   344→          <div className="sticky top-0 z-10 -mx-1 bg-background/95 px-1 pb-2 pt-1 backdrop-blur-sm">
   345→        <div className="flex items-center gap-2">
   346→          <div className="custom-scrollbar flex flex-1 gap-2 overflow-x-auto pb-1" role="tablist" aria-label="壁纸分类">
   347→            {ZHEFENG_CATEGORIES.map((c) => (
   348→              <button
   349→                key={c.key}
   350→                type="button"
   351→                role="tab"
   352→                aria-selected={catKey === c.key}
   353→                onClick={() => setCatKey(c.key)}
   354→                className={cn(
   355→                  "shrink-0 rounded-full border px-4 py-1.5 text-xs font-semibold transition-all",
   356→                  catKey === c.key
   357→                    ? "border-primary bg-primary text-primary-foreground shadow-md shadow-primary/20"
   358→                    : "border-border bg-background text-muted-foreground hover:border-primary/50 hover:text-foreground"
   359→                )}
   360→              >
   361→                {c.name}
   362→              </button>
   363→            ))}
   364→          </div>
   365→
   366→          {/* 已下载过滤开关 */}
   367→          <div className="flex shrink-0 flex-col items-end gap-0.5 pb-1">
   368→            <button
   369→              type="button"
   370→              role="switch"
   371→              aria-checked={hideDownloaded}
   372→              onClick={() => setHideDownloaded((v) => !v)}
   373→              className={cn(
   374→                "inline-flex h-7 items-center gap-1.5 rounded-full border px-2.5 text-[11px] font-semibold transition-colors",
   375→                hideDownloaded
   376→                  ? "border-primary/50 bg-primary/10 text-primary"
   377→                  : "border-border text-muted-foreground hover:text-foreground"
   378→              )}
   379→            >
   380→              <EyeOff className="size-3" />
   381→              隐藏已下载{downloadedCount > 0 && `（${downloadedCount}）`}
   382→            </button>
   383→            {hideDownloaded && downloadedCount > 0 && (
   384→              <button
   385→                type="button"
   386→                onClick={() => void clearDownloads()}
   387→                className="inline-flex h-5.5 items-center gap-0.5 rounded-full px-1.5 text-[10px] font-medium text-muted-foreground transition-colors hover:text-destructive"
   388→              >
   389→                <Trash2 className="size-2.5" />
   390→                清空记录
   391→              </button>
   392→            )}
   393→          </div>
   394→        </div>
   395→      </div>
   396→
   397→      {/* 壁纸网格（无限滚动） */}
   398→      <div
   399→        ref={scrollRef}
   400→        className="custom-scrollbar max-h-[calc(100vh-300px)] min-h-96 overflow-y-auto overscroll-contain pr-1"
   401→      >
   402→        {showSkeleton ? (
   403→          <div className="grid grid-cols-2 gap-3 sm:grid-cols-3 lg:grid-cols-4">
   404→            {Array.from({ length: 8 }).map((_, i) => (
   405→              <div
   406→                key={i}
   407→                className={cn(
   408→                  "animate-pulse rounded-2xl bg-muted",
   409→                  isPhone ? "aspect-[9/16]" : "aspect-[16/10]"
   410→                )}
   411→              />
   412→            ))}
   413→          </div>
   414→        ) : list.error && list.items.length === 0 ? (
   415→          <div className="flex min-h-64 flex-col items-center justify-center gap-3 rounded-2xl border border-dashed p-8 text-center">
   416→            <p className="text-sm font-semibold text-destructive">{list.error}</p>
   417→            <button
   418→              type="button"
   419→              onClick={() => setReloadNonce((n) => n + 1)}
   420→              className="inline-flex h-9 items-center gap-1.5 rounded-full border px-4 text-xs font-semibold transition-colors hover:bg-muted"
   421→            >
   422→              <RefreshCw className="size-3.5" />
   423→              重新抓取
   424→            </button>
   425→          </div>
   426→        ) : list.items.length === 0 ? (
   427→          <div className="flex min-h-64 flex-col items-center justify-center gap-3 text-center">
   428→            <p className="text-sm text-muted-foreground">
   429→              {hideDownloaded && downloadedCount > 0 ? "这一页的壁纸都下载过了" : "这个分类暂时没有壁纸"}
   430→            </p>
   431→            {hideDownloaded && downloadedCount > 0 && (
   432→              <button
   433→                type="button"
   434→                onClick={() => setHideDownloaded(false)}
   435→                className="inline-flex h-8 items-center gap-1.5 rounded-full border px-3 text-xs font-semibold transition-colors hover:bg-muted"
   436→              >
   437→                <EyeOff className="size-3.5" />
   438→                显示已下载的壁纸
   439→              </button>
   440→            )}
   441→          </div>
   442→        ) : (
   443→          <div className={cn("grid gap-3", isPhone ? "grid-cols-2 sm:grid-cols-4 lg:grid-cols-5 xl:grid-cols-6" : "grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 xl:grid-cols-4")}>
   444→            {list.items.map((w) => {
   445→              const order = selected.findIndex((p) => p.id === w.id)
   446→              const isSelected = order >= 0
   447→              return (
   448→                <motion.button
   449→                  key={w.id}
   450→                  type="button"
   451→                  layout
   452→                  initial={{ opacity: 0, y: 12 }}
   453→                  animate={{ opacity: 1, y: 0 }}
   454→                  transition={{ duration: 0.25 }}
   455→                  onClick={() => toggle(w)}
   456→                  aria-pressed={isSelected}
   457→                  aria-label={`壁纸：${w.title}`}
   458→                  className={cn(
   459→                    "group relative overflow-hidden rounded-2xl border-2 text-left transition-all",
   460→                    isPhone ? "aspect-[9/16]" : "aspect-[16/10]",
   461→                    isSelected
   462→                      ? "border-primary shadow-lg shadow-primary/25"
   463→                      : "border-transparent shadow-sm hover:-translate-y-1 hover:shadow-lg"
   464→                  )}
   465→                >
   466→                  {/* eslint-disable-next-line @next/next/no-img-element */}
   467→                  <img
   468→                    src={proxyImageUrl(w.thumb)}
   469→                    alt={`壁纸 ${w.title}`}
   470→                    loading="lazy"
   471→                    decoding="async"
   472→                    className="size-full bg-muted object-cover transition-transform duration-300 group-hover:scale-105"
   473→                  />
   474→
   475→                  {/* 分辨率徽标 */}
