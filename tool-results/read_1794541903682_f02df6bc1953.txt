    89→/** 当前浏览器是否支持自选本机目录 */
    90→function supportsDirectoryPicker(): boolean {
    91→  return (
    92→    typeof window !== "undefined" &&
    93→    typeof (window as unknown as { showDirectoryPicker?: unknown }).showDirectoryPicker === "function"
    94→  )
    95→}

    97→/** 目录选择结果：成功 / 用户取消 / 被环境拦截（iframe 嵌入、无用户激活等） */
    98→type PickDirResult =
    99→  | { kind: "ok"; dir: FsDirHandleLike }
   100→  | { kind: "canceled" }
   101→  | { kind: "blocked"; reason: string }
   102→
   103→/** 弹出浏览器目录选择器（需用户手势触发）；区分「取消」与「被拦截」 */
   104→async function pickWebDirectory(): Promise<PickDirResult> {
   105→  if (!supportsDirectoryPicker()) return { kind: "canceled" }
   106→  try {
   107→    const picker = (
   108→      window as unknown as {
   109→        showDirectoryPicker: (options?: { mode?: "read" | "readwrite" }) => Promise<FsDirHandleLike>
   110→      }
   111→    ).showDirectoryPicker
   112→    const dir = await picker({ mode: "readwrite" })
   113→    return { kind: "ok", dir }
   114→  } catch (e) {
   115→    // AbortError = 用户在系统对话框里点了取消，静默返回
   116→    if (e instanceof DOMException && e.name === "AbortError") return { kind: "canceled" }
   117→    // 其余（SecurityError / 嵌入 iframe 拦截 / 需要用户激活…）都是「没弹出来」→ 需要给用户明确指引
   118→    return {
   119→      kind: "blocked",
   120→      reason: e instanceof Error ? `${e.name}: ${e.message}` : String(e),
   121→    }
   122→  }
   123→}

   125→/** 目录内重名自动追加序号：name (1).ext */
   126→async function uniqueFsName(dir: FsDirHandleLike, filename: string): Promise<string> {
   127→  const dot = filename.lastIndexOf(".")
   128→  const stem = dot === -1 ? filename : filename.slice(0, dot)
   129→  const ext = dot === -1 ? "" : filename.slice(dot)
   130→  let candidate = filename
   131→  for (let i = 0; i < 999; i++) {
   132→    try {
   133→      await dir.getFileHandle(candidate, { create: false }) // 已存在 → 试下一个
   134→      candidate = i === 0 ? `${stem} (1)${ext}` : `${stem} (${i + 1})${ext}`
   135→    } catch {
   136→      return candidate // 不存在 → 可用
   137→    }
   138→  }
   139→  return `${stem}-${Date.now()}${ext}`
   140→}

   142→/** 经自选目录直写文件（成功 true；权限丢失 / 失败 false → 调用方回退浏览器下载） */
   143→async function saveToDirectory(dir: FsDirHandleLike, w: ZhefengWallpaper): Promise<boolean> {
   144→  try {
   145→    if (dir.queryPermission) {
   146→      const p = await dir.queryPermission({ mode: "readwrite" })
   147→      if (p !== "granted" && dir.requestPermission) {
   148→        const r = await dir.requestPermission({ mode: "readwrite" })
   149→        if (r !== "granted") return false
   150→      }
   151→    }
   152→    const res = await fetch(downloadUrl(w), { cache: "no-store" })
   153→    if (!res.ok) throw new Error(`HTTP ${res.status}`)
   154→    const blob = await res.blob()
   155→    // 文件名优先取服务端 Content-Disposition（与桌面版一致的清洗名）
   156→    const cd = res.headers.get("content-disposition") ?? ""
   157→    const m = /filename\*=UTF-8''([^;]+)/.exec(cd)
   158→    const dot = w.src.lastIndexOf(".")
   159→    const fallbackExt = dot === -1 ? "png" : w.src.slice(dot + 1)
   160→    const filename = m
   161→      ? decodeURIComponent(m[1])
   162→      : `${w.title.replace(/[\\/:*?"<>|]/g, " ") || "wallpaper"}.${fallbackExt}`
   163→    const fh = await dir.getFileHandle(await uniqueFsName(dir, filename), { create: true })
   164→    const stream = await fh.createWritable()
   165→    await stream.write(blob)
   166→    await stream.close()
   167→    return true
   168→  } catch (e) {
   169→    console.warn("[workshop] 写入自选目录失败，回退浏览器下载:", e)
   170→    return false
   171→  }
   172→}
