    63→/* ---------- 网页版自选保存目录（File System Access API，Chrome / Edge 支持） ---------- */

    65→/** 最小类型面：仅用到目录写入能力，避免依赖完整 FSA 类型定义 */
    66→interface FsWritable {
    67→  write(data: Blob): Promise<void>
    68→  close(): Promise<void>
    69→}
    70→interface FsFileHandle {
    71→  createWritable(): Promise<FsWritable>
    72→}
    73→interface FsDirHandle {
    74→  readonly name: string
    75→  getFileHandle(name: string, options?: { create?: boolean }): Promise<FsFileHandle>
    76→  queryPermission?(descriptor?: { mode?: "read" | "readwrite" }): Promise<PermissionState>
    77→  requestPermission?(descriptor?: { mode?: "read" | "readwrite" }): Promise<PermissionState>
    78→}

    80→/** 当前浏览器是否支持自选本机目录 */
    81→function supportsDirectoryPicker(): boolean {
    82→  return (
    83→    typeof window !== "undefined" &&
    84→    typeof (window as unknown as { showDirectoryPicker?: unknown }).showDirectoryPicker === "function"
    85→  )
    86→}
    87→
    88→/** 目录选择结果：成功 / 用户取消 / 被环境拦截（iframe 嵌入、无用户激活等） */
    89→type PickDirResult =
    90→  | { kind: "ok"; dir: FsDirHandle }
    91→  | { kind: "canceled" }
    92→  | { kind: "blocked"; reason: string }
    93→
    94→/** 弹出浏览器目录选择器（需用户手势触发）；区分「取消」与「被拦截」 */
    95→async function pickWebDirectory(): Promise<PickDirResult> {
    96→  if (!supportsDirectoryPicker()) return { kind: "canceled" }
    97→  try {
    98→    const picker = (
    99→      window as unknown as {
   100→        showDirectoryPicker: (options?: { mode?: "read" | "readwrite" }) => Promise<FsDirHandle>
   101→      }
   102→    ).showDirectoryPicker
   103→    const dir = await picker({ mode: "readwrite" })
   104→    return { kind: "ok", dir }
   105→  } catch (e) {
   106→    // AbortError = 用户在系统对话框里点了取消，静默返回
   107→    if (e instanceof DOMException && e.name === "AbortError") return { kind: "canceled" }
   108→    // 其余（SecurityError / 嵌入 iframe 拦截 / 需要用户激活…）都是「没弹出来」→ 需要给用户明确指引
   109→    return {
   110→      kind: "blocked",
   111→      reason: e instanceof Error ? `${e.name}: ${e.message}` : String(e),
   112→    }
   113→  }
   114→}

   116→/** 目录内重名自动追加序号：name (1).ext */
   117→async function uniqueFsName(dir: FsDirHandle, filename: string): Promise<string> {
   118→  const dot = filename.lastIndexOf(".")
   119→  const stem = dot === -1 ? filename : filename.slice(0, dot)
   120→  const ext = dot === -1 ? "" : filename.slice(dot)
   121→  let candidate = filename
   122→  for (let i = 0; i < 999; i++) {
   123→    try {
   124→      await dir.getFileHandle(candidate, { create: false }) // 已存在 → 试下一个
   125→      candidate = i === 0 ? `${stem} (1)${ext}` : `${stem} (${i + 1})${ext}`
   126→    } catch {
   127→      return candidate // 不存在 → 可用
   128→    }
   129→  }
   130→  return `${stem}-${Date.now()}${ext}`
   131→}

   133→/** 经自选目录直写文件（成功 true；权限丢失 / 失败 false → 调用方回退浏览器下载） */
   134→async function saveToDirectory(dir: FsDirHandle, w: ZhefengWallpaper): Promise<boolean> {
   135→  try {
   136→    if (dir.queryPermission) {
   137→      const p = await dir.queryPermission({ mode: "readwrite" })
   138→      if (p !== "granted" && dir.requestPermission) {
   139→        const r = await dir.requestPermission({ mode: "readwrite" })
   140→        if (r !== "granted") return false
   141→      }
   142→    }
   143→    const res = await fetch(downloadUrl(w), { cache: "no-store" })
   144→    if (!res.ok) throw new Error(`HTTP ${res.status}`)
   145→    const blob = await res.blob()
   146→    // 文件名优先取服务端 Content-Disposition（与桌面版一致的清洗名）
   147→    const cd = res.headers.get("content-disposition") ?? ""
   148→    const m = /filename\*=UTF-8''([^;]+)/.exec(cd)
   149→    const dot = w.src.lastIndexOf(".")
   150→    const fallbackExt = dot === -1 ? "png" : w.src.slice(dot + 1)
   151→    const filename = m
   152→      ? decodeURIComponent(m[1])
   153→      : `${w.title.replace(/[\\/:*?"<>|]/g, " ") || "wallpaper"}.${fallbackExt}`
   154→    const fh = await dir.getFileHandle(await uniqueFsName(dir, filename), { create: true })
   155→    const stream = await fh.createWritable()
   156→    await stream.write(blob)
   157→    await stream.close()
   158→    return true
   159→  } catch (e) {
   160→    console.warn("[workshop] 写入自选目录失败，回退浏览器下载:", e)
   161→    return false
   162→  }
   163→}
