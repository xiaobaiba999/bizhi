    66→    66→interface FsWritable {

    88→    88→/** 弹出浏览器目录选择器（需用户手势触发），取消 / 被拦截时返回 null */
    89→    89→async function pickWebDirectory(): Promise<FsDirHandle | null> {
    90→    90→  if (!supportsDirectoryPicker()) return null
    91→    91→  try {
    92→    92→    const picker = (
    93→    93→      window as unknown as {
    94→    94→        showDirectoryPicker: (options?: { mode?: "read" | "readwrite" }) => Promise<FsDirHandle>
    95→    95→      }
    96→    96→    ).showDirectoryPicker
    97→    97→    return await picker({ mode: "readwrite" })
    98→    98→  } catch {
    99→    99→    return null
   100→   100→  }
   101→   101→}

   103→   103→/** 目录内重名自动追加序号：name (1).ext */
   104→   104→async function uniqueFsName(dir: FsDirHandle, filename: string): Promise<string> {
   105→   105→  const dot = filename.lastIndexOf(".")
   106→   106→  const stem = dot === -1 ? filename : filename.slice(0, dot)
   107→   107→  const ext = dot === -1 ? "" : filename.slice(dot)
   108→   108→  let candidate = filename
   109→   109→  for (let i = 0; i < 999; i++) {
   110→   110→    try {
   111→   111→      await dir.getFileHandle(candidate, { create: false }) // 已存在 → 试下一个
   112→   112→      candidate = i === 0 ? `${stem} (1)${ext}` : `${stem} (${i + 1})${ext}`
   113→   113→    } catch {
   114→   114→      return candidate // 不存在 → 可用
   115→   115→    }
   116→   116→  }
   117→   117→  return `${stem}-${Date.now()}${ext}`
   118→   118→}

   120→   120→/** 经自选目录直写文件（成功 true；权限丢失 / 失败 false → 调用方回退浏览器下载） */
   121→   121→async function saveToDirectory(dir: FsDirHandle, w: ZhefengWallpaper): Promise<boolean> {
   122→   122→  try {
   123→   123→    if (dir.queryPermission) {
   124→   124→      const p = await dir.queryPermission({ mode: "readwrite" })
   125→   125→      if (p !== "granted" && dir.requestPermission) {
   126→   126→        const r = await dir.requestPermission({ mode: "readwrite" })
   127→   127→        if (r !== "granted") return false
   128→   128→      }
   129→   129→    }
   130→   130→    const res = await fetch(downloadUrl(w), { cache: "no-store" })
   131→   131→    if (!res.ok) throw new Error(`HTTP ${res.status}`)
   132→   132→    const blob = await res.blob()
   133→   133→    // 文件名优先取服务端 Content-Disposition（与桌面版一致的清洗名）
   134→   134→    const cd = res.headers.get("content-disposition") ?? ""
   135→   135→    const m = /filename\*=UTF-8''([^;]+)/.exec(cd)
   136→   136→    const dot = w.src.lastIndexOf(".")
   137→   137→    const fallbackExt = dot === -1 ? "png" : w.src.slice(dot + 1)
   138→   138→    const filename = m
   139→   139→      ? decodeURIComponent(m[1])
   140→   140→      : `${w.title.replace(/[\\/:*?"<>|]/g, " ") || "wallpaper"}.${fallbackExt}`
   141→   141→    const fh = await dir.getFileHandle(await uniqueFsName(dir, filename), { create: true })
   142→   142→    const stream = await fh.createWritable()
   143→   143→    await stream.write(blob)
   144→   144→    await stream.close()
   145→   145→    return true
   146→   146→  } catch (e) {
   147→   147→    console.warn("[workshop] 写入自选目录失败，回退浏览器下载:", e)
   148→   148→    return false
   149→   149→  }
   150→   150→}
