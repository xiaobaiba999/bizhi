    66→/** 最小类型面：仅用到目录写入能力，避免依赖完整 FSA 类型定义 */
    67→interface FsWritable {
    68→  write(data: Blob): Promise<void>
    69→  close(): Promise<void>
    70→}
    71→interface FsFileHandle {
    72→  createWritable(): Promise<FsWritable>
    73→}
    74→interface FsDirHandle {
    75→  readonly name: string
    76→  getFileHandle(name: string, options?: { create?: boolean }): Promise<FsFileHandle>
    77→  queryPermission?(descriptor?: { mode?: "read" | "readwrite" }): Promise<PermissionState>
    78→  requestPermission?(descriptor?: { mode?: "read" | "readwrite" }): Promise<PermissionState>
    79→}

    81→/** 当前浏览器是否支持自选本机目录 */
    82→function supportsDirectoryPicker(): boolean {
    83→  return (
    84→    typeof window !== "undefined" &&
    85→    typeof (window as unknown as { showDirectoryPicker?: unknown }).showDirectoryPicker === "function"
    86→  )
    87→}
    88→
    89→/** 目录选择结果：成功 / 用户取消 / 被环境拦截（iframe 嵌入、无用户激活等） */
    90→type PickDirResult =
    91→  | { kind: "ok"; dir: FsDirHandle }
    92→  | { kind: "canceled" }
    93→  | { kind: "blocked"; reason: string }
    94→
    95→/** 弹出浏览器目录选择器（需用户手势触发）；区分「取消」与「被拦截」 */
    96→async function pickWebDirectory(): Promise<PickDirResult> {
    97→  if (!supportsDirectoryPicker()) return { kind: "canceled" }
    98→  try {
    99→    const picker = (
   100→      window as unknown as {
   101→        showDirectoryPicker: (options?: { mode?: "read" | "readwrite" }) => Promise<FsDirHandle>
   102→      }
   103→    ).showDirectoryPicker
   104→    const dir = await picker({ mode: "readwrite" })
   105→    return { kind: "ok", dir }
   106→  } catch (e) {
   107→    // AbortError = 用户在系统对话框里点了取消，静默返回
   108→    if (e instanceof DOMException && e.name === "AbortError") return { kind: "canceled" }
   109→    // 其余（SecurityError / 嵌入 iframe 拦截 / 需要用户激活…）都是「没弹出来」→ 需要给用户明确指引
   110→    return {
   111→      kind: "blocked",
   112→      reason: e instanceof Error ? `${e.name}: ${e.message}` : String(e),
   113→    }
   114→  }
   115→}

   117→/** 目录内重名自动追加序号：name (1).ext */
   118→async function uniqueFsName(dir: FsDirHandle, filename: string): Promise<string> {
   119→  const dot = filename.lastIndexOf(".")
   120→  const stem = dot === -1 ? filename : filename.slice(0, dot)
   121→  const ext = dot === -1 ? "" : filename.slice(dot)
   122→  let candidate = filename
   123→  for (let i = 0; i < 999; i++) {
   124→    try {
   125→      await dir.getFileHandle(candidate, { create: false }) // 已存在 → 试下一个
   126→      candidate = i === 0 ? `${stem} (1)${ext}` : `${stem} (${i + 1})${ext}`
   127→    } catch {
   128→      return candidate // 不存在 → 可用
   129→    }
   130→  }
   131→  return `${stem}-${Date.now()}${ext}`
   132→}

   134→/** 经自选目录直写文件（成功 true；权限丢失 / 失败 false → 调用方回退浏览器下载） */
   135→async function saveToDirectory(dir: FsDirHandle, w: ZhefengWallpaper): Promise<boolean> {
   136→  try {
   137→    if (dir.queryPermission) {
   138→      const p = await dir.queryPermission({ mode: "readwrite" })
   139→      if (p !== "granted" && dir.requestPermission) {
   140→        const r = await dir.requestPermission({ mode: "readwrite" })
   141→        if (r !== "granted") return false
   142→      }
   143→    }
   144→    const res = await fetch(downloadUrl(w), { cache: "no-store" })
   145→    if (!res.ok) throw new Error(`HTTP ${res.status}`)
   146→    const blob = await res.blob()
   147→    // 文件名优先取服务端 Content-Disposition（与桌面版一致的清洗名）
   148→    const cd = res.headers.get("content-disposition") ?? ""
   149→    const m = /filename\*=UTF-8''([^;]+)/.exec(cd)
   150→    const dot = w.src.lastIndexOf(".")
   151→    const fallbackExt = dot === -1 ? "png" : w.src.slice(dot + 1)
   152→    const filename = m
   153→      ? decodeURIComponent(m[1])
   154→      : `${w.title.replace(/[\\/:*?"<>|]/g, " ") || "wallpaper"}.${fallbackExt}`
   155→    const fh = await dir.getFileHandle(await uniqueFsName(dir, filename), { create: true })
   156→    const stream = await fh.createWritable()
   157→    await stream.write(blob)
   158→    await stream.close()
   159→    return true
   160→  } catch (e) {
   161→    console.warn("[workshop] 写入自选目录失败，回退浏览器下载:", e)
   162→    return false
   163→  }
   164→}
